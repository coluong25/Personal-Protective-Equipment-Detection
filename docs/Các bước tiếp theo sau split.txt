1. Verify Split Data
Trước khi train, kiểm tra lại split có đúng tỉ lệ và không bị leak không.

import os
from pathlib import Path

def verify_split(data_root: str):
    splits = ["train", "val", "test"]
    for split in splits:
        img_dir = Path(data_root) / split / "images"
        lbl_dir = Path(data_root) / split / "labels"
        
        imgs = list(img_dir.glob("*"))
        lbls = list(lbl_dir.glob("*.txt"))
        
        # Kiểm tra số ảnh và label phải khớp nhau
        print(f"{split}: {len(imgs)} images, {len(lbls)} labels")
        assert len(imgs) == len(lbls), f"Mismatch in {split}!"

verify_split("datasets/helmet")

3. Chọn model và Train
Với helmet detection, YOLOv8 là lựa chọn phổ biến nhất hiện tại.


from ultralytics import YOLO

# Load pretrained model (transfer learning)
model = YOLO("yolov8s.pt")  # s = small, cân bằng tốc độ/độ chính xác

model.train(
    data="helmet.yaml",
    epochs=50,
    imgsz=640,
    batch=16,
    patience=10,        # early stopping nếu không cải thiện sau 10 epoch
    device=0,           # GPU 0, dùng "cpu" nếu không có GPU
    project="runs/helmet",
    name="exp1",
)

4. Evaluate model

# Đánh giá trên tập val/test
metrics = model.val(data="helmet.yaml", split="test")

# Các chỉ số quan trọng cần xem
print(f"mAP50:     {metrics.box.map50:.3f}")   # mean Average Precision @ IoU 0.5
print(f"mAP50-95:  {metrics.box.map:.3f}")     # mAP trung bình nhiều ngưỡng IoU
print(f"Precision: {metrics.box.mp:.3f}")
print(f"Recall:    {metrics.box.mr:.3f}")


5. Inference thực tế

# Test trên ảnh mới
results = model.predict(
    source="test_images/",
    conf=0.5,           # ngưỡng confidence tối thiểu
    iou=0.45,           # ngưỡng NMS (Non-Maximum Suppression)
    save=True,          # lưu ảnh kết quả
)
